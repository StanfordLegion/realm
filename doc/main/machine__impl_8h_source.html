246 // The machine query cache.
247 //
248 // A query result is an immutable, sorted list stamped with the generation
249 // of the cache it was built for (the generation is bumped by every machine
250 // update). Lists live in snapshots: a snapshot holds the lists for one
251 // generation and is itself immutable once published. Readers take the
252 // current snapshot with one atomic load, so a query that the cache can
253 // serve writes no shared memory at all; this is what lets concurrent
254 // queries scale. A query remembers the list it last used (see
255 // ProcessorQueryImpl::attached), which is still valid while its generation
256 // is current, and each thread remembers where its last step through a list
257 // stopped, so iterating a cached list in order costs a few comparisons per
258 // step.
259 //
260 // A replaced snapshot is retired, not freed, until cleanup_query_caches()
261 // at runtime shutdown, because a query may still hold a pointer into it
262 // (attached_list() dereferences the pointer to compare generations). So
263 // every machine update that is followed by a query retains one snapshot of
264 // each kind, each a copy of the machine's lists. That is bounded today
265 // because the machine model only changes at startup. Snapshot builds are
266 // single-flight (QueryCacheSlot::get_or_build): an update costs one build
267 // per snapshot kind, not one per querying thread.
268 //
269 // Before the machine model may change at run time (elasticity), retired
270 // snapshots need reclaiming. What would have to change: a query would keep
271 // the generation of its attached list beside the pointer (one 64-bit atomic
272 // holding the generation and an index into a table of per-generation
273 // snapshot headers, say), so that staleness is decided without touching
274 // the snapshot; then only a call in progress can be inside a retired
275 // snapshot, and since calls are short and never block while reading one,
276 // an epoch or quiescent-state scheme (readers announce the generation they
277 // entered; a snapshot is freed once every reader has moved past it) can
278 // free retired snapshots safely. Until then the retention is by design.
279 template <typename T>
280 struct QueryResultList {
281 unsigned generation = 0;
282 std::vector<T> items; // sorted by id
283 };

294 // the publication slot for one kind of snapshot
295 template <typename S>
296 class QueryCacheSlot {
297 public:
298 QueryCacheSlot(void)
299 : current(nullptr)
300 , builds(0)
301 {}
302 // the snapshots are freed by cleanup_query_caches() at runtime shutdown,
303 // not by this object's destructor: a process that exits without shutting
304 // Realm down may still have threads querying, and a hit on a leaked
305 // snapshot is harmless where a freed one is not. (The slot's own members
306 // are destroyed with the other statics at exit, like every Realm static;
307 // a thread still building a snapshot at that point is an application
308 // bug, as it was before.)
309
310 // the published snapshot if it is for 'generation', else null
311 const S *get(unsigned generation) const
312 {
313 const S *s = current.load_acquire();
314 return (s && (s->generation == generation)) ? s : nullptr;
315 }
316
317 // the snapshot for 'generation', built by 'build(generation)' if it is
318 // missing. Builders are serialized: after a machine update the first
319 // thread to miss builds, and the others wait for its snapshot instead of
320 // each building their own, so an update costs one build however many
321 // threads query. Returns null if the live generation has moved on from
322 // 'generation' (the caller then walks the machine), whether before the
323 // build started or while it ran. A build is attempted only for the live
324 // generation and generations only move forward, so each generation is
325 // built at most once. Lock order: build_mutex, then the machine's reader
326 // lock inside 'build', then 'mutex' in install(); machine writers take
327 // neither of this class's locks
328 template <typename BuildFn>
329 const S *get_or_build(unsigned generation, const atomic<unsigned int> &live,
330 BuildFn &&build)
331 {
332 AutoLock<UnfairMutex> al(build_mutex);
333 const S *s = get(generation);
334 if(s)
335 return s; // built while we waited
336 if(live.load() != generation)
337 return nullptr;
338 builds.fetch_add(1);
339 S *built = build(generation);
340 // the live generation is read after the build (call arguments are not
341 // sequenced), so a change during the build is seen by install()
342 return install(built, live.load());
343 }
344
345 // publishes 'built' (taking ownership) and returns it - unless a snapshot
346 // for the same generation got there first, which is returned instead, or
347 // the cache generation has moved on from 'built', in which case null is
348 // returned (and the caller walks the machine)
349 const S *install(S *built, unsigned live_generation);
350
351 // frees every snapshot ever published (runtime shutdown)
352 void cleanup(void);
353
354 // how many snapshots have been built (for tests)
355 unsigned long build_count(void) const { return builds.load(); }
356
357 protected:
358 atomic<const S *> current;
359 Mutex mutex; // serializes install and cleanup
360 std::vector<const S *> retired;
361 UnfairMutex build_mutex; // serializes builders (see get_or_build)
362 atomic<unsigned long> builds;
363 };
