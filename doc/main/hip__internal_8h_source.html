38 do { \
39 hipError_t ret = (cmd); \
40 if(ret != hipSuccess) { \
41 fprintf(stderr, "HIP: %s = %d (%s)\n", #cmd, ret, hipGetErrorString(ret)); \

49 const char *name, *str; \
